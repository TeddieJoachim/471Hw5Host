        def dxds(r,s): return (2 + r + 0.2*sin(5*pi*s))*(-sin(0.5*pi*s)*(0.5*pi))  +  cos(0.5*pi*s)*(0.2*cos(5*pi*s)*(5*pi))

        return dxdr(r,s)*dyds(r,s) - dxds(r,s)*dydr(r,s)
